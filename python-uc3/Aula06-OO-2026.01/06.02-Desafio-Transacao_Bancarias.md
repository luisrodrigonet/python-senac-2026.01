## :one: Arquivo: `conta.py`

Iniciamos o código com a importação dos módulos, e com a definição do valor inicial para o número da conta corrente. Em seguida declaramos a classe `ContaBancaria` e seus atributos.



```python
from datetime import datetime

# Gerador de número de conta sequencial
conta_sequencial = 1000

class ContaBancaria:
    def __init__(self, titular, saldo_inicial=0, limite_saque=500):
        global conta_sequencial
        self.numero = conta_sequencial
        conta_sequencial += 1

        self.titular = titular
        self.__saldo = saldo_inicial
        self.limite_saque = limite_saque
        self.historico = []
        self.saques_diarios = 0
        self.data_ultimo_saque = None
```

```python

    def depositar(self, valor):
        if valor > 0:
            self.__saldo += valor
            self._registrar_transacao(f"Depósito: +R$ {valor:.2f}")
            print(f"✅ Depósito de R$ {valor:.2f} realizado com sucesso.")
        else:
            print("❌ Valor de depósito inválido.")

    def sacar(self, valor):
        hoje = datetime.now().date()

        if self.data_ultimo_saque != hoje:
            self.saques_diarios = 0

        if valor > self.limite_saque:
            print(f"❌ O valor excede o limite diário de saque de R$ {self.limite_saque:.2f}.")
        elif valor > self.__saldo:
            print("❌ Saldo insuficiente.")
        elif self.saques_diarios >= 3:
            print("❌ Limite de 3 saques diários atingido.")
        elif valor <= 0:
            print("❌ Valor de saque inválido.")
        else:
            self.__saldo -= valor
            self.saques_diarios += 1
            self.data_ultimo_saque = hoje
            self._registrar_transacao(f"Saque: -R$ {valor:.2f}")
            print(f"✅ Saque de R$ {valor:.2f} realizado com sucesso.")

    def transferir(self, outra_conta, valor):
        if valor > 0 and valor <= self.__saldo:
            self.__saldo -= valor
            outra_conta.__saldo += valor
            self._registrar_transacao(f"Transferência para {outra_conta.titular}: -R$ {valor:.2f}")
            outra_conta._registrar_transacao(f"Recebido de {self.titular}: +R$ {valor:.2f}")
            print(f"✅ Transferência de R$ {valor:.2f} para {outra_conta.titular} realizada.")
        elif valor > self.__saldo:
            print("❌ Saldo insuficiente para transferência.")
        else:
            print("❌ Valor de transferência inválido.")

    def mostrar_saldo(self):
        print(f"🏦 Saldo de {self.titular}: R$ {self.__saldo:.2f}")

    def mostrar_historico(self):
        print(f"\n📜 Histórico de transações de {self.titular} (Conta #{self.numero}):")
        if not self.historico:
            print("ℹ️ Nenhuma transação registrada.")
        for transacao in self.historico:
            print(f" - {transacao}")
        print()

    def _registrar_transacao(self, descricao):
        agora = datetime.now().strftime("%d/%m/%Y %H:%M:%S")
        self.historico.append(f"{agora} - {descricao}")

    def get_saldo(self):
        return self.__saldo


class ContaPoupanca(ContaBancaria):
    def __init__(self, titular, saldo_inicial=0, juros=0.005):  # Juros mensais de 0.5%
        super().__init__(titular, saldo_inicial)
        self.juros = juros  # Taxa mensal

    def aplicar_juros(self):
        juros = self.get_saldo() * self.juros
        self.depositar(juros)
        self._registrar_transacao(f"Juros aplicados: +R$ {juros:.2f}")
        print(f"💰 Juros de R$ {juros:.2f} aplicados à conta de {self.titular}.")
```



## :two: Arquivo: `main.py`

```python
from conta import ContaBancaria, ContaPoupanca

def menu():
    print("\n=== Sistema Bancário ===")
    print("1. Depositar")
    print("2. Sacar")
    print("3. Transferir")
    print("4. Mostrar saldo")
    print("5. Mostrar histórico")
    print("6. Aplicar juros (Poupança)")
    print("0. Sair")
    return input("Escolha uma opção: ")

def selecionar_conta(contas):
    print("\nContas disponíveis:")
    for i, conta in enumerate(contas):
        tipo = "Poupança" if isinstance(conta, ContaPoupanca) else "Corrente"
        print(f"{i} - {conta.titular} (#{conta.numero}) - {tipo}")
    try:
        idx = int(input("Selecione uma conta pelo número: "))
        return contas[idx]
    except (ValueError, IndexError):
        print("❌ Conta inválida.")
        return None

def main():
    # Criando contas iniciais
    conta_joao = ContaBancaria("João", 1000)
    conta_maria = ContaPoupanca("Maria", 500)
    contas = [conta_joao, conta_maria]

    while True:
        print("\n=== Menu Principal ===")
        print("1. Selecionar conta")
        print("2. Listar contas")
        print("0. Sair")
        opcao = input("Escolha uma opção: ")

        if opcao == "0":
            print("👋 Saindo do sistema. Até logo!")
            break

        elif opcao == "1":
            conta = selecionar_conta(contas)
            if conta:
                while True:
                    escolha = menu()
                    if escolha == "0":
                        break

                    elif escolha == "1":
                        valor = float(input("Digite o valor do depósito: "))
                        conta.depositar(valor)

                    elif escolha == "2":
                        valor = float(input("Digite o valor do saque: "))
                        conta.sacar(valor)

                    elif escolha == "3":
                        outra_conta = selecionar_conta(contas)
                        if outra_conta and outra_conta != conta:
                            valor = float(input("Digite o valor da transferência: "))
                            conta.transferir(outra_conta, valor)

                    elif escolha == "4":
                        conta.mostrar_saldo()

                    elif escolha == "5":
                        conta.mostrar_historico()

                    elif escolha == "6":
                        if isinstance(conta, ContaPoupanca):
                            conta.aplicar_juros()
                        else:
                            print("❌ Esta conta não é do tipo Poupança.")

                    else:
                        print("❌ Opção inválida.")

        elif opcao == "2":
            print("\nContas cadastradas:")
            for conta in contas:
                tipo = "Poupança" if isinstance(conta, ContaPoupanca) else "Corrente"
                print(f"- {conta.titular} (#{conta.numero}) - {tipo}")
        else:
            print("❌ Opção inválida.")

if __name__ == "__main__":
    main()
```
